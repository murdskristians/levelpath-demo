# Plan: Light/dark theme toggle

> Spec: agents/78d872a3/spec.md · Type: feature

## Approach

A small `theme.ts` (modelled on `users.ts`) loads, saves and applies the theme. An inline script in `index.html` sets `data-theme` on `<html>` before first paint; React state starts from that attribute, so they agree. `App.css` colours move into custom properties on `:root` (light, unchanged values) and `:root[data-theme='dark']`. A `ThemeToggle` button sits beside `UserPicker` in the shared `picker` node, so both header states get it. Chosen over `prefers-color-scheme`-only CSS because the user must be able to override the OS.

## Steps

### 1. `theme.ts`

```ts
export type Theme = 'light' | 'dark';
export const THEME_KEY = 'levelpath.theme';

export function systemTheme(): Theme {
  return typeof window.matchMedia === 'function' &&
    window.matchMedia('(prefers-color-scheme: dark)').matches
    ? 'dark'
    : 'light';
}

export function loadTheme(): Theme {
  try {
    const v = localStorage.getItem(THEME_KEY);
    if (v === 'light' || v === 'dark') return v;
  } catch (err) {
    console.warn('theme: could not read the stored theme', err);
  }
  return systemTheme();
}

export function saveTheme(theme: Theme): void {
  try {
    localStorage.setItem(THEME_KEY, theme);
  } catch (err) {
    console.warn('theme: could not store the theme', err);
  }
}

export function applyTheme(theme: Theme): void {
  document.documentElement.dataset.theme = theme;
}
```

`loadTheme` is used for the initial state (it recomputes the same answer as the inline script; if `data-theme` is already `light|dark`, prefer it: `const attr = document.documentElement.dataset.theme; if (attr === 'light' || attr === 'dark') return attr;` at the top of a separate `initialTheme()` that `App` uses). In tests no inline script runs, so `initialTheme()` must fall through to `loadTheme()`. Because `data-theme` persists on `document.documentElement` between tests, tests must `delete document.documentElement.dataset.theme` in `afterEach`.

### 2. `ThemeToggle.tsx`

```tsx
import type { Theme } from '../theme';

export function ThemeToggle({ theme, onToggle }: { theme: Theme; onToggle: () => void }) {
  return (
    <button
      type="button"
      className="theme-toggle secondary"
      aria-pressed={theme === 'dark'}
      onClick={onToggle}
    >
      Dark theme
    </button>
  );
}
```

### 3. `App.tsx`

- `const [theme, setTheme] = useState<Theme>(initialTheme);` and `useEffect(() => applyTheme(theme), [theme]);`.
- `toggleTheme`: `const next = theme === 'dark' ? 'light' : 'dark'; saveTheme(next); setTheme(next);`.
- `const picker = (<div className="header-controls"><UserPicker …/><ThemeToggle theme={theme} onToggle={toggleTheme} /></div>);`. It is still the one node used by both headers. `Dashboard`'s `key={userId}` remount does not touch theme state (it lives in `App`).

### 4. `index.html`

Before the module script, in `<head>`:

```html
<script>
  (function () {
    var theme;
    try {
      var stored = localStorage.getItem('levelpath.theme');
      if (stored === 'light' || stored === 'dark') theme = stored;
    } catch (err) {
      console.warn('theme: could not read the stored theme', err);
    }
    if (!theme) {
      theme =
        window.matchMedia && window.matchMedia('(prefers-color-scheme: dark)').matches
          ? 'dark'
          : 'light';
    }
    document.documentElement.setAttribute('data-theme', theme);
  })();
</script>
```

Keep Prettier happy (`pnpm lint` checks HTML).
